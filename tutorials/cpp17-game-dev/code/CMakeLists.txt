cmake_minimum_required(VERSION 3.16)

project(cpp17_game_dev_tutorial LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(MSVC)
    add_compile_options(/W4 /permissive-)
elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-Wall -Wextra -Wpedantic)
endif()

enable_testing()
add_subdirectory(01-modern-syntax-and-raii)
add_subdirectory(02-move-semantics-and-smart-pointers)
add_subdirectory(03-sequence-containers)
add_subdirectory(04-associative-containers)
add_subdirectory(05-iterators-algorithms)
add_subdirectory(06-game-entity-storage)
add_subdirectory(07-templates)
add_subdirectory(08-compile-time)
add_subdirectory(09-threads)
add_subdirectory(10-atomics)
add_subdirectory(11-async-thread-pool)
add_subdirectory(12-game-loop)
add_subdirectory(13-ecs)
add_subdirectory(14-game-patterns)
add_subdirectory(15-lifecycle-handles)
add_subdirectory(16-job-system)
